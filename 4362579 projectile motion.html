import numpy as np
import matplotlib.pyplot as plt
from scipy.integrate import solve_ivp

# Constants
g = 9.81  
v0 = 34.72  
angles_deg = [30, 45]
angles = [np.radians(a) for a in angles_deg]

# Parameters for drag (Q2)
m = 0.029  
Cd = 0.1   
rho = 1.225  
A = 0.01   
b = 0.5 * Cd * rho * A  


# Q1: No Air Resistance
def projectile_no_drag(angle):
    vx0 = v0 * np.cos(angle)
    vy0 = v0 * np.sin(angle)

    # time of flight
    t_flight = 2 * vy0 / g
    t = np.linspace(0, t_flight, 300)

    # equations of motion
    x = vx0 * t
    y = vy0 * t - 0.5 * g * t**2

    # horizontal distance
    R = vx0 * t_flight
    return x, y, R


plt.figure(figsize=(8,5))
for ang, ang_deg in zip(angles, angles_deg):
    x, y, R = projectile_no_drag(ang)
    plt.plot(x, y, label=f"{ang_deg}°  (Range={R:.2f} m)")
    print(f"Q1: Distance travelled at {ang_deg}° = {R:.2f} m")

plt.xlabel("Horizontal Distance (m)")
plt.ylabel("Vertical Height (m)")
plt.title("Projectile Motion (No Air Resistance)")
plt.legend()
plt.grid()
plt.show()



# Q2: With Air Resistance
def equations(t, u):
    x, y, vx, vy = u
    v = np.sqrt(vx**2 + vy**2)
    ax = -(b/m) * v * vx
    ay = -g - (b/m) * v * vy
    return [vx, vy, ax, ay]

def projectile_with_drag(angle):
    vx0 = v0 * np.cos(angle)
    vy0 = v0 * np.sin(angle)
    u0 = [0, 0, vx0, vy0]

    t_span = (0, 10)
    t_eval = np.linspace(0, 10, 500)

    sol = solve_ivp(equations, t_span, u0, t_eval=t_eval)

    x, y = sol.y[0], sol.y[1]

    # stop when projectile hits the ground
    ground_idx = np.where(y >= 0)[0]
    x, y = x[ground_idx], y[ground_idx]
    R = x[-1]
    return x, y, R


plt.figure(figsize=(8,5))
for ang, ang_deg in zip(angles, angles_deg):
    x, y, R = projectile_with_drag(ang)
    plt.plot(x, y, label=f"{ang_deg}°  (Range={R:.2f} m)")
    print(f"Q2: Distance travelled at {ang_deg}° = {R:.2f} m")

plt.xlabel("Horizontal Distance (m)")
plt.ylabel("Vertical Height (m)")
plt.title("Projectile Motion (With Air Resistance)")
plt.legend()
plt.grid()
plt.show()
